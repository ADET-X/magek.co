export async function onRequestPost(context) { try { const formData = await context.request.formData(); const file = formData.get('file'); if (!file) { return new Response(JSON.stringify({ success: false, error: 'File video tidak ditemukan' }), { status: 400, headers: { 'Content-Type': 'application/json' } }); } // Persiapan data sesuai dokumentasi API Catbox const catboxForm = new FormData(); catboxForm.append('reqtype', 'fileupload'); catboxForm.append('fileToUpload', file); // userhash tidak diisi agar upload menjadi anonymous // Kirim ke server Catbox const response = await fetch('https://catbox.moe/user/api.php', { method: 'POST', body: catboxForm }); const resultText = await response.text(); // Catbox mengembalikan URL teks biasa jika sukses (contoh: https://files.catbox.moe/xyz123.mp4) if (resultText.startsWith('http')) { const fileName = resultText.split('/').pop(); return new Response(JSON.stringify({ success: true, filecode: fileName }), { headers: { 'Content-Type': 'application/json' } }); } else { return new Response(JSON.stringify({ success: false, error: 'Catbox menolak: ' + resultText }), { status: 500, headers: { 'Content-Type': 'application/json' } }); } } catch (err) { return new Response(JSON.stringify({ success: false, error: 'Sistem Error: ' + err.message }), { status: 500, headers: { 'Content-Type': 'application/json' } }); } }
