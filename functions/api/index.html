export async function onRequestPost(context) { try { // 1. Dapatkan request asli dari browser const request = context.request; // 2. Karena browser sudah mengirim request sebagai FormData murni, // kita bisa langsung meneruskan (passthrough) body tersebut ke Catbox // tanpa perlu membongkarnya di memori Cloudflare. const response = await fetch('https://catbox.moe/user/api.php', { method: 'POST', headers: { // Teruskan header khusus (terutama Content-Type boundary multipart/form-data) 'Content-Type': request.headers.get('Content-Type') }, // Gunakan body mentah (streaming) agar RAM Cloudflare tidak penuh body: request.body, // Trik untuk Cloudflare Workers/Pages agar bisa streaming body duplex: 'half' }); const resultText = await response.text(); // 3. Catbox mengembalikan URL jika sukses if (resultText.startsWith('http')) { const fileName = resultText.split('/').pop(); return new Response(JSON.stringify({ success: true, filecode: fileName }), { headers: { 'Content-Type': 'application/json' } }); } else { return new Response(JSON.stringify({ success: false, error: 'Catbox menolak: ' + resultText }), { status: 500, headers: { 'Content-Type': 'application/json' } }); } } catch (err) { return new Response(JSON.stringify({ success: false, error: 'Sistem Error: ' + err.message }), { status: 500, headers: { 'Content-Type': 'application/json' } }); } }
